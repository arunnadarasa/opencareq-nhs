from ehrql import codelist_from_csv, create_dataset, days, minimum_of
from ehrql.tables.tpp import (
    addresses, apcs, clinical_events, medications, patients, practice_registrations
)

dataset = create_dataset()
dataset.configure_dummy_data(population_size=12000)

smr_codes = codelist_from_csv("codelists/nhse-structured-medication-review-snomed.csv", column="code")
sp_referral_codes = codelist_from_csv("codelists/nhse-dapb4066-social-prescribing-referral.csv", column="code")
cyp3a4_doac_dmd = codelist_from_csv("codelists/opencareq-cyp3a4-doac-sglt2i-dmd.csv", column="code")
acute_instability_excl = codelist_from_csv("codelists/opencareq-dcb0129-h01-acute-excl.csv", column="code")
nco_fracture_codes = codelist_from_csv("codelists/opencareq-nco-extremity-fracture.csv", column="code")

first_smr = (
    clinical_events.where(clinical_events.snomedct_code.is_in(smr_codes))
    .where(clinical_events.date.is_on_or_between("2025-04-01", "2026-03-31"))
    .sort_by(clinical_events.date).first_for_patient()
)
index_date = first_smr.date
dataset.define_population(
    first_smr.exists_for_patient()
    & (patients.age_on(index_date) >= 18)
    & practice_registrations.for_patient_on(index_date).exists_for_patient()
    & medications.where(medications.dmd_code.is_in(cyp3a4_doac_dmd)
        & medications.date.is_on_or_between(index_date - days(90), index_date)).exists_for_patient()
    & (~clinical_events.where(clinical_events.snomedct_code.is_in(acute_instability_excl)
        & clinical_events.date.is_on_or_between(index_date - days(7), index_date)).exists_for_patient())
)
sp_ref = clinical_events.where(clinical_events.snomedct_code.is_in(sp_referral_codes)).where(
    clinical_events.date.is_on_or_between(index_date, index_date + days(180))
).sort_by(clinical_events.date).first_for_patient()
dataset.initiated_within_28d_grace = sp_ref.date.is_on_or_between(index_date, index_date + days(28))
dataset.imd_quintile = addresses.for_patient_on(index_date).imd_quintile
